# Only two examples: the minimal application and the successive Game of Life stages.
include(Frontends)

add_executable(HelloWorld hello-world/HelloWorld.cpp)
target_link_libraries(HelloWorld PRIVATE WebFront WebFront_warnings WebFront_options)
webfront_setup_frontends_for_target(HelloWorld)

# The simulation can be explored before adding a UI or enabling a native engine.
add_executable(WebFrontLifeConsole life/LifeConsole.cpp)
target_link_libraries(WebFrontLifeConsole PRIVATE WebFront_warnings WebFront_options)

add_executable(WebFrontLifeDeveloper life/LifeDeveloper.cpp)
target_link_libraries(WebFrontLifeDeveloper PRIVATE WebFront WebFront_warnings WebFront_options)
webfront_setup_frontends_for_target(WebFrontLifeDeveloper)

set(WEBFRONT_LIFE_ASSETS "" CACHE FILEPATH "LifeUI.hpp generated by wf embed (enables the offline Life example)")
if(WEBFRONT_LIFE_ASSETS)
    if(NOT EXISTS "${WEBFRONT_LIFE_ASSETS}")
        message(FATAL_ERROR "WEBFRONT_LIFE_ASSETS must name an existing LifeUI.hpp generated by wf embed")
    endif()
    add_executable(WebFrontLifeEmbedded life/LifeEmbedded.cpp)
    target_link_libraries(WebFrontLifeEmbedded PRIVATE WebFront WebFront_warnings WebFront_options)
    target_compile_definitions(WebFrontLifeEmbedded PRIVATE WEBFRONT_LIFE_ASSETS_HEADER="${WEBFRONT_LIFE_ASSETS}")
    webfront_setup_frontends_for_target(WebFrontLifeEmbedded)
endif()
