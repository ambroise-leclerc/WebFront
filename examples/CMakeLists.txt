# Only two examples: the minimal application and the successive Game of Life stages.
include(Frontends)

# The Life examples use the native window when its engine is built and otherwise fall back to the system browser,
# which needs no dependency: a missing engine is explained here, never an error.
webfront_native_frontend_option(WEBFRONT_LIFE_NATIVE_OPTION)
set(WEBFRONT_LIFE_NATIVE_WINDOW OFF)
if(WEBFRONT_LIFE_NATIVE_OPTION)
    if(${WEBFRONT_LIFE_NATIVE_OPTION})
        set(WEBFRONT_LIFE_NATIVE_WINDOW ON)
    endif()
endif()
if(NOT WEBFRONT_LIFE_NATIVE_WINDOW)
    if(NOT WEBFRONT_LIFE_NATIVE_OPTION)
        set(_life_install "no native engine is registered for ${CMAKE_SYSTEM_NAME}")
    elseif(NOT WEBFRONT_AUTODETECT_NATIVE)
        set(_life_install "enable it with -D${WEBFRONT_LIFE_NATIVE_OPTION}=ON (automatic detection is off)")
    elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        set(_life_install "install the WebKitGTK 6.0 development package (libwebkitgtk-6.0-dev on Ubuntu/Debian, webkitgtk6.0-devel on Fedora; see docs/WEBKITGTK.md)")
    elseif(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
        set(_life_install "use Clang on macOS 13.3 or later")
    else()
        set(_life_install "use an MSVC-compatible compiler")
    endif()
    if(WEBFRONT_LIFE_NATIVE_OPTION)
        set(_life_advice "For a native window, ${_life_install}, then run cmake again.")
    else()
        set(_life_advice "${_life_install}; use the default browser on this platform.")
    endif()
    message(NOTICE
        "Native window: no native engine will be built.\n"
        "  The Life examples will open the default browser instead (frontend DefaultBrowser).\n"
        "  ${_life_advice}\n"
        "  CEF (-DWEBFRONT_EMBED_CEF=ON) makes CEFFrontend available to your own applications; the examples do not select it.")
endif()
# Select the same frontend for the developer and embedded Life stages.
function(webfront_life_frontend target)
    if(WEBFRONT_LIFE_NATIVE_WINDOW)
        target_compile_definitions(${target} PRIVATE WEBFRONT_LIFE_NATIVE_WINDOW)
    endif()
endfunction()

add_executable(HelloWorld hello-world/HelloWorld.cpp)
target_link_libraries(HelloWorld PRIVATE WebFront WebFront_warnings WebFront_options)
webfront_setup_frontends_for_target(HelloWorld)

# The simulation can be explored before adding a UI or enabling a native engine.
add_executable(WebFrontLifeConsole life/LifeConsole.cpp)
target_link_libraries(WebFrontLifeConsole PRIVATE WebFront_warnings WebFront_options)

add_executable(WebFrontLifeDeveloper life/LifeDeveloper.cpp)
target_link_libraries(WebFrontLifeDeveloper PRIVATE WebFront WebFront_warnings WebFront_options)
webfront_setup_frontends_for_target(WebFrontLifeDeveloper)
webfront_life_frontend(WebFrontLifeDeveloper)

add_executable(WebFrontLifeReact life/react/LifeReact.cpp)
target_link_libraries(WebFrontLifeReact PRIVATE WebFront WebFront_warnings WebFront_options)
webfront_setup_frontends_for_target(WebFrontLifeReact)
webfront_life_frontend(WebFrontLifeReact)

set(WEBFRONT_LIFE_ASSETS "" CACHE FILEPATH "LifeUI.hpp generated by wf embed (enables the offline Life example)")
if(WEBFRONT_LIFE_ASSETS)
    if(NOT EXISTS "${WEBFRONT_LIFE_ASSETS}")
        message(FATAL_ERROR "WEBFRONT_LIFE_ASSETS must name an existing LifeUI.hpp generated by wf embed")
    endif()
    add_executable(WebFrontLifeEmbedded life/LifeEmbedded.cpp)
    target_link_libraries(WebFrontLifeEmbedded PRIVATE WebFront WebFront_warnings WebFront_options)
    target_compile_definitions(WebFrontLifeEmbedded PRIVATE WEBFRONT_LIFE_ASSETS_HEADER="${WEBFRONT_LIFE_ASSETS}")
    webfront_setup_frontends_for_target(WebFrontLifeEmbedded)
    webfront_life_frontend(WebFrontLifeEmbedded)
endif()
