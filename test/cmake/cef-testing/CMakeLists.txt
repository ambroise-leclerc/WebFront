cmake_minimum_required(VERSION 3.31)
project(CEFTestingScope NONE)
list(APPEND CMAKE_MODULE_PATH "${ROOT}/cmake")
include("${CPM_MODULE}")
set(CPM_CEF_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/dependency")
# Simulate Apple's branch without a native compiler or SDK; no source mutation is allowed.
set(APPLE "${SIMULATE_APPLE}")
set(MSVC OFF)
unset(CEF_ROOT)
unset(BUILD_TESTING)
unset(BUILD_TESTING CACHE)
if(STATE MATCHES "^cache-(ON|OFF|empty)$")
    set(initial "${CMAKE_MATCH_1}")
    if(initial STREQUAL "empty")
        set(initial "")
    endif()
    set(BUILD_TESTING "${initial}" CACHE BOOL "Fixture testing setting")
elseif(STATE MATCHES "^normal-(ON|OFF|empty)$")
    set(initial "${CMAKE_MATCH_1}")
    if(initial STREQUAL "empty")
        set(initial "")
    endif()
    set(BUILD_TESTING "${initial}")
elseif(STATE STREQUAL "normal-ON-cache-OFF")
    set(initial ON)
    set(BUILD_TESTING OFF CACHE BOOL "Fixture testing setting")
    set(BUILD_TESTING ON)
elseif(STATE STREQUAL "normal-OFF-cache-ON")
    set(initial OFF)
    set(BUILD_TESTING ON CACHE BOOL "Fixture testing setting")
    set(BUILD_TESTING OFF)
elseif(NOT STATE STREQUAL "undefined")
    message(FATAL_ERROR "Unknown state '${STATE}'")
endif()
set(had_cache OFF)
if(DEFINED CACHE{BUILD_TESTING})
    set(had_cache ON)
    get_property(initial_cache CACHE BUILD_TESTING PROPERTY VALUE)
endif()
include("${ROOT}/cmake/CEFIntegration.cmake")
foreach(target cef_sanity_test cef_resources_test cef_window_test)
    get_target_property(excluded ${target} EXCLUDE_FROM_ALL)
    if(NOT excluded)
        message(FATAL_ERROR "CEF demo ${target} must be excluded from the default build")
    endif()
endforeach()
if(EXISTS "${CMAKE_BINARY_DIR}/_deps/cef-src" OR EXISTS "${CMAKE_BINARY_DIR}/_deps/cef_binaries-src")
    message(FATAL_ERROR "CEF setup must not write SDK patches or wrapper stubs into _deps")
endif()
get_property(configured GLOBAL PROPERTY test_cef_configured)
if(NOT configured)
    message(FATAL_ERROR "The local dependency was not configured by CPM")
endif()
if(STATE STREQUAL "undefined")
    if(DEFINED BUILD_TESTING)
        message(FATAL_ERROR "Originally undefined BUILD_TESTING must remain undefined")
    endif()
elseif(NOT DEFINED BUILD_TESTING OR NOT BUILD_TESTING STREQUAL initial)
    message(FATAL_ERROR "Expected BUILD_TESTING='${initial}', got '${BUILD_TESTING}'")
endif()
if(had_cache)
    get_property(actual_cache CACHE BUILD_TESTING PROPERTY VALUE)
    get_property(actual_help CACHE BUILD_TESTING PROPERTY HELPSTRING)
    if(NOT DEFINED CACHE{BUILD_TESTING} OR NOT actual_cache STREQUAL initial_cache
       OR NOT actual_help STREQUAL "Fixture testing setting")
        message(FATAL_ERROR "CEF modified the caller's BUILD_TESTING cache entry")
    endif()
elseif(DEFINED CACHE{BUILD_TESTING})
    message(FATAL_ERROR "CEF created a BUILD_TESTING cache entry")
endif()
