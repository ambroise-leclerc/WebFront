cmake_minimum_required(VERSION 3.31)
project(FrontendConsumer LANGUAGES CXX)

# Register and detect inside a child directory, then use the API from its parent and a sibling.
add_subdirectory(provider)
include("${ROOT}/cmake/Frontends.cmake")
function(check_consumer target)
    webfront_native_frontend_option(option)
    if(NOT option STREQUAL "WEBFRONT_EMBED_CONSUMER")
        message(FATAL_ERROR "Consumer lost native option: '${option}'")
    endif()
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${target}.cpp" "int main() { return 0; }\n")
    add_executable(${target} "${CMAKE_CURRENT_BINARY_DIR}/${target}.cpp")
    webfront_setup_frontends_for_target(${target})
    get_target_property(other_configured ${target} CONSUMER_OTHER_SETUP)
    if(NOT MUTATE_OTHER AND NOT other_configured)
        message(FATAL_ERROR "General frontend setup lost the other enabled engine")
    elseif(MUTATE_OTHER AND other_configured)
        message(FATAL_ERROR "Consumer setup ignored the support module's final option")
    endif()
    get_target_property(configured ${target} CONSUMER_SETUP)
    if(PROBE AND NOT configured)
        message(FATAL_ERROR "Consumer setup: expected ${PROBE}, got '${configured}'")
    endif()
    if(NOT PROBE AND configured)
        message(FATAL_ERROR "Disabled engine setup was called")
    endif()
    add_executable(${target}_native "${CMAKE_CURRENT_BINARY_DIR}/${target}.cpp")
    webfront_use_native_window(${target}_native)
    get_target_property(other_configured ${target}_native CONSUMER_OTHER_SETUP)
    if(other_configured)
        message(FATAL_ERROR "Native helper must not set up other embedded engines")
    endif()
    get_target_property(configured ${target}_native CONSUMER_SETUP)
    if(PROBE AND NOT configured)
        message(FATAL_ERROR "Native helper did not apply the engine's setup")
    elseif(NOT PROBE AND configured)
        message(FATAL_ERROR "Native helper configured a disabled engine")
    endif()
    get_target_property(definitions ${target}_native COMPILE_DEFINITIONS)
    if(PROBE AND NOT "WEBFRONT_NATIVE_WINDOW" IN_LIST definitions)
        message(FATAL_ERROR "Consumer lost native window definition")
    elseif(NOT PROBE AND "WEBFRONT_NATIVE_WINDOW" IN_LIST definitions)
        message(FATAL_ERROR "Browser fallback defines native window")
    endif()
endfunction()
check_consumer(parent_app)
add_subdirectory(sibling)

if(PROBE)
    function(check_override)
        set(WEBFRONT_EMBED_CONSUMER OFF)
        add_executable(override_app "${CMAKE_CURRENT_BINARY_DIR}/parent_app.cpp")
        webfront_use_native_window(override_app)
        get_target_property(definitions override_app COMPILE_DEFINITIONS)
        if("WEBFRONT_NATIVE_WINDOW" IN_LIST definitions)
            message(FATAL_ERROR "An explicit caller variable must override the published result")
        endif()
    endfunction()
    check_override()
endif()

function(check_unsupported)
    set(CMAKE_SYSTEM_NAME UnsupportedOS)
    add_executable(unsupported_app "${CMAKE_CURRENT_BINARY_DIR}/parent_app.cpp")
    webfront_use_native_window(unsupported_app)
    get_target_property(definitions unsupported_app COMPILE_DEFINITIONS)
    if("WEBFRONT_NATIVE_WINDOW" IN_LIST definitions)
        message(FATAL_ERROR "Unsupported platform must use the browser fallback")
    endif()
endfunction()
check_unsupported()
