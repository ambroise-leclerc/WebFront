cmake_minimum_required(VERSION 3.31)
project(FrontendWiring NONE)
enable_testing()
include("${ROOT}/cmake/Frontends.cmake")
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(browser IMPORTED)
set_target_properties(browser PROPERTIES IMPORTED_LOCATION "${CMAKE_COMMAND}")

set(CMAKE_SYSTEM_NAME Linux)
set(WEBFRONT_EMBED_CEF ON)
set(WEBFRONT_EMBED_WEBVIEW2 OFF)
set(frontend CEF)
set(name WebFrontBrowserIntegration)
set(expected_timeout 60)
set(expect_xvfb OFF)
if(CASE STREQUAL "extension" OR CASE STREQUAL "missing-xvfb-hook")
    webfront_register_frontend(TestEmbedded
        OPTION WEBFRONT_EMBED_TESTEMBEDDED POLICY strict DESCRIPTION "Test engine"
        PLATFORMS Linux SUPPORT TestSupport SETUP setup_test_engine
        TEST_PLATFORMS Linux TEST_NAME TestEmbeddedIntegration TEST_TIMEOUT 42 XVFB_LINUX TEST_SETUP test_engine_lifecycle)
    set(WEBFRONT_EMBED_CEF OFF)
    webfront_declare_frontend_options()
    if(WEBFRONT_EMBED_TESTEMBEDDED)
        message(FATAL_ERROR "New engines must default to OFF")
    endif()
    set(WEBFRONT_EMBED_TESTEMBEDDED ON)
    webfront_include_frontend_support()
    if(DEFINED _webfront_name OR DEFINED _webfront_option)
        message(FATAL_ERROR "Support loading leaked temporary variables")
    endif()
    if(NOT test_engine_support_loaded)
        message(FATAL_ERROR "The enabled support module was not loaded in the caller scope")
    endif()
    webfront_default_embedded_frontend(frontend FALLBACK CEF)
    webfront_select_frontend(SELECTED "${frontend}" "Test selection" EMBEDDED_ONLY REQUIRE_AVAILABLE)
    if(NOT SELECTED_TYPE STREQUAL "webfront::frontend::TestEmbeddedFrontend")
        message(FATAL_ERROR "The new engine was not selected")
    endif()
    webfront_setup_frontends_for_target(browser)
    get_target_property(configured browser TEST_ENGINE_CONFIGURED)
    if(NOT configured)
        message(FATAL_ERROR "The new engine's target setup was not called")
    endif()
    set(name TestEmbeddedIntegration)
    set(expected_timeout 42)
    if(CASE STREQUAL "extension")
        set(expect_xvfb ON)
    endif()
elseif(CASE STREQUAL "cef-linux")
    set(expect_xvfb ON)
elseif(CASE STREQUAL "cef-macos")
    set(CMAKE_SYSTEM_NAME Darwin)
elseif(CASE STREQUAL "cef-windows")
    set(CMAKE_SYSTEM_NAME Windows)
elseif(CASE STREQUAL "cef-disabled")
    set(WEBFRONT_EMBED_CEF OFF)
elseif(CASE STREQUAL "browser")
    set(frontend DefaultBrowser)
elseif(CASE STREQUAL "webview2-windows")
    set(CMAKE_SYSTEM_NAME Windows)
    set(WEBFRONT_EMBED_CEF OFF)
    set(WEBFRONT_EMBED_WEBVIEW2 ON)
    set(frontend WebView2)
    set(name WebFrontWebView2Integration)
    set(expected_timeout 90)
    function(webfront_add_webview2_tests)
        add_test(NAME EngineLifecycle COMMAND "${CMAKE_COMMAND}" -E true)
    endfunction()
endif()
if(CASE STREQUAL "missing-xvfb" OR CASE STREQUAL "missing-xvfb-hook")
    # Discovery is deliberately unavailable even with a stale cache and an arbitrary prefix.
    set(XVFB_RUN_EXECUTABLE "${CMAKE_COMMAND}" CACHE FILEPATH "Stale discovery" FORCE)
    set(CMAKE_PREFIX_PATH "${ROOT}")
    function(webfront_find_xvfb out)
        set(${out} "XVFB_RUN_EXECUTABLE-NOTFOUND" PARENT_SCOPE)
    endfunction()
endif()
if(expect_xvfb)
    # A deterministic stand-in: this fixture checks generated commands, never runs the browser.
    set(XVFB_RUN_EXECUTABLE "${CMAKE_COMMAND}" CACHE FILEPATH "Xvfb fixture" FORCE)
endif()
webfront_add_browser_tests(browser "${frontend}" "${ROOT}/webtest")
if(CASE STREQUAL "missing-xvfb-hook")
    if(TEST ${name} OR NOT TEST EngineLifecycle)
        message(FATAL_ERROR "Missing Xvfb must skip Jasmine while preserving independent engine tests")
    endif()
    return()
endif()
if(CASE MATCHES "^(cef-disabled|missing-xvfb|cef-macos|cef-windows|browser)$")
    if(TEST ${name})
        message(FATAL_ERROR "An unavailable browser test must not be registered")
    endif()
    return()
endif()
set(expected_TIMEOUT "${expected_timeout}")
set(expected_LABELS web-integration)
set(expected_RESOURCE_LOCK webfront_port_9002)
set(expected_WORKING_DIRECTORY "${ROOT}/webtest")
if(NOT EXISTS "${expected_WORKING_DIRECTORY}/JasmineTest.cpp")
    message(FATAL_ERROR "Expected the actual WebFront browser asset directory")
endif()
foreach(property TIMEOUT LABELS RESOURCE_LOCK WORKING_DIRECTORY)
    get_test_property(${name} ${property} actual)
    if(NOT actual STREQUAL expected_${property})
        message(FATAL_ERROR "${name} ${property}: expected '${expected_${property}}', got '${actual}'")
    endif()
endforeach()
if(CASE STREQUAL "extension" AND NOT TEST EngineLifecycle)
    message(FATAL_ERROR "The new engine's lifecycle tests were not registered")
endif()
